### 2. 重试逻辑 (Retry with Exponential Backoff)
**当前问题：**
- 下载失败直接跳过，不重试
- 临时网络问题导致大量资源丢失

**改进方案：**
```javascript
async downloadWithRetry(url, maxRetries = 3) {
  let lastError;
  for (let attempt = 1; attempt <= maxRetries; attempt++) {
    try {
      return await this.fetcher.fetch(url);
    } catch (error) {
      lastError = error;
      if (attempt < maxRetries) {
        const delay = this.config.download.retryDelay * Math.pow(2, attempt - 1);
        logger.warn(`Retry ${attempt}/${maxRetries} for ${url} after ${delay}ms`);
        await new Promise(resolve => setTimeout(resolve, delay));
      }
    }
  }
  throw lastError;
}
```

**实现位置：** `controller/crawl_controller.js` (新增方法，processQueue中调用)


### 3. 资源提取 (Resource Extraction from HTML)
**当前问题：**
- 只下载HTML链接，CSS/JS/图片/字体等资源被忽略
- 下载的HTML页面无法正常显示

**改进方案：**
```javascript
// 在 HtmlParse 或新建 ResourceExtractor 类
extractResources(html, baseUrl) {
  const resources = [];
  const $ = cheerio.load(html);

  // CSS files
  $('link[rel="stylesheet"]').each((_, el) => {
    const href = $(el).attr('href');
    if (href) resources.push({ url: href, type: 'css' });
  });

  // JavaScript files
  $('script[src]').each((_, el) => {
    const src = $(el).attr('src');
    if (src) resources.push({ url: src, type: 'js' });
  });

  // Images
  $('img[src], source[srcset]').each((_, el) => {
    const src = $(el).attr('src') || $(el).attr('srcset');
    if (src) resources.push({ url: src, type: 'image' });
  });

  // Fonts (in CSS @font-face)
  // Videos/Audio
  $('video source, audio source').each((_, el) => {
    const src = $(el).attr('src');
    if (src) resources.push({ url: src, type: 'media' });
  });

  return resources;
}
```

**实现位置：** `services/html_parse.js` 或新建 `services/resource_extractor.js`


### 4. URL过滤 (URL Filtering)
**当前问题：**
- 无排除模式，下载管理后台/API等不需要的内容
- 浪费带宽和存储空间

**改进方案：**
```javascript
// 在 DomainContext.isAllowedUrl()
isAllowedUrl(url) {
  const urlStr = url.toString();

  // Check exclude patterns
  for (const pattern of this.excludePatterns) {
    if (pattern.test(urlStr)) {
      return false;
    }
  }

  // Check include patterns (if any)
  if (this.includePatterns.length > 0) {
    let matched = false;
    for (const pattern of this.includePatterns) {
      if (pattern.test(urlStr)) {
        matched = true;
        break;
      }
    }
    if (!matched) return false;
  }

  return true;
}
```

**实现位置：** `libs/domain_context.js` (在 resolveHref 中调用)


### 5. 进度统计 (Statistics Tracking)
**当前问题：**
- 只有简单日志，无法了解整体进度
- 不知道还需要多久完成

**改进方案：**
```javascript
class Statistics {
  constructor() {
    this.startTime = Date.now();
    this.totalQueued = 0;
    this.totalProcessed = 0;
    this.totalFailed = 0;
    this.totalBytes = 0;
    this.byType = {
      html: 0, css: 0, js: 0, images: 0, other: 0
    };
  }

  record(url, success, bytes, contentType) {
    if (success) {
      this.totalProcessed++;
      this.totalBytes += bytes;
      this.recordByType(contentType);
    } else {
      this.totalFailed++;
    }
  }

  getProgress() {
    const elapsed = Date.now() - this.startTime;
    const rate = this.totalProcessed / (elapsed / 1000);
    return {
      processed: this.totalProcessed,
      failed: this.totalFailed,
      pending: this.totalQueued - this.totalProcessed - this.totalFailed,
      rate: rate.toFixed(2) + ' pages/sec',
      totalMB: (this.totalBytes / 1024 / 1024).toFixed(2),
      elapsed: (elapsed / 1000).toFixed(0) + 's'
    };
  }
}
```

**实现位置：** 新建 `libs/statistics.js`


### 6. 并发下载 (Concurrent Downloads)
**当前问题：**
- 串行下载，速度很慢
- 1000个页面可能需要几小时

**改进方案：**
```javascript
async processQueueConcurrent() {
  const concurrency = this.config.limits.maxConcurrent || 5;
  const workers = [];

  for (let i = 0; i < concurrency; i++) {
    workers.push(this.worker(i));
  }

  await Promise.all(workers);
}

async worker(id) {
  while (this.queue.hasPending()) {
    const item = this.queue.dequeue();
    if (!item) break;

    await this.processItem(item);

    // Rate limiting
    if (this.config.download.delay > 0) {
      await new Promise(r => setTimeout(r, this.config.download.delay));
    }
  }
}
```

**实现位置：** `controller/crawl_controller.js` (替换当前的 processQueue)


### 7. 断点续传 (Resume Support)
**当前问题：**
- 中断后需要完全重新开始
- Ctrl+C 会丢失所有进度

**改进方案：**
```javascript
// 定期保存进度
async saveProgress() {
  const state = {
    processed: Array.from(this.queue.processed),
    pending: this.queue.pending,
    statistics: this.statistics.getProgress()
  };
  await fs.promises.writeFile(
    'progress.json',
    JSON.stringify(state, null, 2)
  );
}

// 加载进度
async loadProgress() {
  try {
    const data = await fs.promises.readFile('progress.json', 'utf8');
    const state = JSON.parse(data);
    this.queue.processed = new Set(state.processed);
    this.queue.pending = state.pending;
    return true;
  } catch (error) {
    return false;
  }
}
```

**实现位置：** `controller/crawl_controller.js`
