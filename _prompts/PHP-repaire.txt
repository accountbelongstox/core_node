# Bridge the libxml2 SONAME gap so Sury PHP 8.5 can install AND run on bleeding-edge
# Debian derivatives (e.g. Kali rolling).
#
# Sury's php8.5-cli / php8.5-xml depend on the historic `libxml2` package, which
# ships libxml2.so.2. Rolling derivativ already moved to
# libxml2 2.15 (package `libxml2-16`, soname libxml2.so.16) and dropped the old
# `libxml2` package with no virtual Pro unsatisfiable
# and, even if force-installed, the PHP binary (linked against .so.2) would fail to
# load. libxml2 is core, non-disableable PHP, so this blocks PHP entirely.
#
# When the gap is detected, install Debian's `libxml2` (.so.2) compat package for the
# resolved Sury codename (the suite Sury built against, so the ABI matches). It depends
# only on libc6/liblzma5/zlib1g (present on such systems) and co-installs cleanly next
# to libxml2-16 (different package name, different .so files). Idempotent; a no-op when
# libxml2.so.2 is already present or the distro still ships a real `libxml2` candidate.
#
# Opt out with PHP_LIBXML2_COMPAT_SHIM=0 -- then a clear gap message is printed and the
# caller's PHP install is left to fail loudly instead of silently mixing in a lib.
ensure_php_libxml2_compat_from_apt_repository_manager() {
    local php_vendor="$1"     # ubuntu | debian
    local php_codename="$2"   # resolved repo codename (e.g. trixie)

    # Only the Debian/Sury path is affected; Ubuntu's ondrej PPA tracks Ubuntu libs.
    [ "$php_vendor" = "debian" ] || return 0

    # Already satisfiable? Either the distro offers a real `libxml2` candidate, the
    # .so.2 is already on disk (prior sge is installed.
    local cand
    cand="$(apt-cache policy libxml2 2>/dev/null | awk '/Candidate:/{print $2}')"
    if [ -n "$cand" ] && [ "$cand" != "(none)" ]; then
        return 0
    fi
    if ldconfig -p 2>/dev/null | grep -q 'libxml2\.so\.2'; then
        return 0
    fi
    if dpkg-query -W -f='${Status}' libxml2 2>/dev/null | grep -q 'install ok installed'; then
        return 0
    fi

    echo "[libxml2-compat] This distro has no 'libxml2' (libxml2.so.2) package that Sury PHP 8.5 needs;" >&2
    echo "[libxml2-compat]   it ships only the newer libxml2-16 / libxml2.so.16." >&2

    case "${PHP_LIBXML2_COMPAT_SHIM:-1}" in
        0|false|no|off)
            echo "[libxml2-compat] PHP_-> NOT installing the compat lib." >&2
            echo "[libxml2-compat] The PHP 8.5 install will fail until libxml2.so.2 is provided." >&2
            echo "[libxml2-compat] Re-rIM=1 (default) to auto-install it." >&2
            return 0
            ;;
    esac

    local arch; arch="$(dpkg --print-architecture 2>/dev/null || echo amd64)"
    local idx_url="http://deb.debian.org/debian/dists/${php_codename}/main/binary-${arch}/Packages.gz"
    echo "[libxml2-compat] Resolving Debian ${php_codename} libxml2 (.so.2) compat package..." >&2

    local tmp; tmp="$(mktemp -d 2>/dev/null || echo "/tmp/libxml2compat.$$")"
    mkdir -p "$tmp"
    local idx="$tmp/Packages.gz"
    if ! curl -fsSL -o "$idx" "$idx_url"; then
        echo "[libxml2-compat] ERROR: could not fetch Debian package index ($idx_url)." >&2
        rm -rf "$tmp"; return 1
    fi

    # Filename: of the binary package named exactly 'libxml2' (the .so.2 one). The
    # trailing \n in the regex preventsxml2-dev / etc.
    local fname
    fname="$(zcat "$idx" 2>/dev/null \
        | awk 'BEGIN{RS="";ORS="\n\n"} /(^|\n)Package: libxml2\n/' \
        | sed -n 's/^Filename: //p' | head -n1)"
    if [ -z "$fname" ]; then
        echo "[libxml2-compat] ERROR: Debian ${php_codename} has no 'libxml2' (.so.2) package." >&2
        rm -rf "$tmp"; return 1
    fi

    local deb_url="http://deb.debian.org/debian/${fname}"
    local deb="$tmp/libxml2-compat.deb"
    echo "[libxml2-compat] Downloading ${deb_url}" >&2
    if ! curl -fsSL -o "$deb" "$deb_url"; then
        echo "[libxml2-compat] ERROR: download failed ($deb_url)." >&2
        rm -rf "$tmp"; return 1
    fi

    echo "[libxml2-compat] Installing compat libxml2 (.so.2) alongside libxml2-16..." >&2
    if ! $USE_SUDO dpkg -i "$deb"; then
        # Pull any missing deps from th1g normally present).
        $USE_SUDO apt-get -f install -y 2>/dev/null || true
    fi
    rm -rf "$tmp"

    if ldconfig -p 2>/dev/null | grep -q 'libxml2\.so\.2'; then
        echo "[libxml2-compat] OK: libxml2.so.2 now available (Debian ${php_codename} compat lib)." >&2
        return 0
    fi
    echo "[libxml2-compat] WARNING: libxml2.so.2 still absent after shim; PHP install may fail." >&2
    return 1
}

# PHP? Returns 0 (yes / undeterminable) or 1 (explicitly excluded). Parses the simple
# ">=X.Y" + optional "<A.B" forms Swoole uses (e.g. ">=8.2 <8.6", ">=8.1 <8.5").
swoole_ref_supports_php() {
    local ref="$1" php_ver="$2"
    local pvid; pvid="$(echo "$php_ver"100+$2}')"
    local constraint
    constraint="$(curl -fsSL "https://rle/swoole-src/${ref}/composer.json" 2>/dev/
null \
        | sed -n 's/.*"php"[[:space:]]*:[[:space:]]*"\([^"]*\)".*/\1/p' | head -n1)"
    [ -z "$constraint" ] && return 0   # offline / unknown -> do not block
    local lo hi
    lo="$(printf '%s' "$constraint" | grep -oE '>=?[[:space:]]*[0-9]+\.[0-9]+' | head -n1 | grep -oE '[0-9]+\.[
0-9]+')"
    hi="$(printf '%s' "$constraint" | grep -oE '<[[:space:]]*[0-9]+\.[0-9]+'   | head -n1 | grep -oE '[0-9]+\.[
0-9]+')"
    if [ -n "$lo" ]; then
        local loid; loid="$(echo "$lo" | awk -F. '{printf "%d", $1*100+$2}')"
        [ "$pvid" -lt "$loid" ] && return 1
    fi
    if [ -n "$hi" ]; then
