6. UNIFIED HTTP CLIENT USAGE
====================================================================================================

Import:
  import { httpClient } from '@/common/utils/http-client';

Example - Server-side Tool:
  // Image Converter
  const convertImage = async (file: File, format: string) => {
    const formData = new FormData();
    formData.append('image', file);
    formData.append('format', format);

    return await httpClient.post('/api/ittools/v1/image/convert', formData, {
      headers: { 'Content-Type': 'multipart/form-data' }
    });
  };

Example - Client-side Tool:
  // Age Calculator (Pure client-side, no API call)
  const calculateAge = (birthDate: Date) => {
    const today = new Date();
    let age = today.getFullYear() - birthDate.getFullYear();
    const monthDiff = today.getMonth() - birthDate.getMonth();
    if (monthDiff < 0 || (monthDiff === 0 && today.getDate() < birthDate.getDate())) {
      age--;
    }
    return age;
  };

Basic Vue Component Structure:

<template>
  <div class="tool-container">
    <div class="tool-header">
      <h2>{{ toolName }}</h2>
      <p class="tool-description">{{ description }}</p>
    </div>

    <div class="tool-content">
      <!-- Input Section -->
      <div class="input-section">
        <!-- Form fields -->
      </div>

      <!-- Output Section -->
      <div class="output-section">
        <!-- Results display -->
      </div>

      <!-- Action Buttons -->
      <div class="actions">
        <button @click="execute" :disabled="loading">
          {{ loading ? 'Processing...' : 'Execute' }}
        </button>
      </div>
    </div>
  </div>
</template>

<script setup lang="ts">
import { ref } from 'vue';
import { httpClient } from '@/common/utils/http-client';

const loading = ref(false);
const result = ref(null);

const execute = async () => {
  loading.value = true;
  try {
    result.value = await httpClient.post('/api/ittools/v1/...', { ... });
  } catch (error) {
    console.error('Tool execution failed:', error);
  } finally {
    loading.value = false;
  }
};
</script>
