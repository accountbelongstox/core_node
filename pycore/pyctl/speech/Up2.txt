```python
from pycore.pyfoundations.third_party import get_third_package_sqlalchemy
sqlalchemy = get_third_package_sqlalchemy()
from pycore.database.base_model import BaseModel
from pycore.database.models.table_keys import TableKeys

class UtilConfigModel(BaseModel):
    """
    Util Config Model - Global configuration storage

    Schema:
        - id: Integer (PK)
        - key: String(255) - Config key (indexed)
        - value: Text - Config value (JSON string)
        - value_type: String(50) - Type hint (string, int, bool, list, dict)
        - namespace: String(100) - Config namespace (speech, ui, etc.)
        - description: Text - Config description
        - created_at: DateTime
        - updated_at: DateTime

    Indexes:
        - Unique constraint on (key, namespace)
        - Index on key
        - Index on namespace
    """

    __table_key__ = TableKeys.UTIL_CONFIG
    __namespace__ = "util_config"
    __table_name__ = "global_config"
    __schema_version__ = 1

    @classmethod
    def define_table_structure(cls, metadata):
        return sqlalchemy.Table(
            cls.__full_table_name__,
            metadata,
            sqlalchemy.Column('id', sqlalchemy.Integer, primary_key=True, autoincrement=True),
            sqlalchemy.Column('key', sqlalchemy.String(255), nullable=False, index=True),
            sqlalchemy.Column('value', sqlalchemy.Text, nullable=False),
            sqlalchemy.Column('value_type', sqlalchemy.String(50), nullable=False, default='string'),
            sqlalchemy.Column('namespace', sqlalchemy.String(100), nullable=False, index=True, default='default'),
            sqlalchemy.Column('description', sqlalchemy.Text, nullable=True),
            sqlalchemy.Column('created_at', sqlalchemy.DateTime, default=datetime.utcnow, nullable=False),
            sqlalchemy.Column('updated_at', sqlalchemy.DateTime, default=datetime.utcnow, onupdate=datetime.utcnow),

            sqlalchemy.UniqueConstraint('key', 'namespace', name='uq_config_key_namespace'),
        )

    @classmethod
    def get_config(cls, conn, key: str, namespace: str = 'default') -> Optional[Any]:
        """Get config value (auto-deserializes JSON)"""
        result = cls.select_one(conn, where={'key': key, 'namespace': namespace})
        if result:
            value = result['value']
            value_type = result.get('value_type', 'string')

            if value_type == 'string':
                return value
            else:
                return json.loads(value)
        return None

    @classmethod
    def set_config(cls, conn, key: str, value: Any, namespace: str = 'default', description: Optional[str] = None):
        """Set config value (auto-serializes to JSON)"""
        # Detect value type
        if isinstance(value, str):
            value_type = 'string'
            value_str = value
        elif isinstance(value, bool):
            value_type = 'bool'
            value_str = json.dumps(value)
        elif isinstance(value, int):
            value_type = 'int'
            value_str = json.dumps(value)
        elif isinstance(value, (list, dict)):
            value_type = 'list' if isinstance(value, list) else 'dict'
            value_str = json.dumps(value)
        else:
            value_type = 'string'
            value_str = str(value)

        # Check if exists
        existing = cls.select_one(conn, where={'key': key, 'namespace': namespace})

        if existing:
            # Update
            update_data = {
                'value': value_str,
                'value_type': value_type,
                'updated_at': datetime.utcnow()
            }
            if description:
                update_data['description'] = description
            cls.update(conn, update_data, where={'key': key, 'namespace': namespace})
        else:
            # Insert
            cls.insert(conn, {
                'key': key,
                'value': value_str,
                'value_type': value_type,
                'namespace': namespace,
                'description': description,
                'created_at': datetime.utcnow(),
                'updated_at': datetime.utcnow()
            })

    @classmethod
    def get_namespace_configs(cls, conn, namespace: str = 'default') -> Dict[str, Any]:
        """Get all configs for a namespace"""
        results = cls.select(conn, where={'namespace': namespace})
        configs = {}
        for result in results:
            value = result['value']
            value_type = result.get('value_type', 'string')

            if value_type == 'string':
                configs[result['key']] = value
            else:
                configs[result['key']] = json.loads(value)

        return configs

    @classmethod
    def delete_config(cls, conn, key: str, namespace: str = 'default') -> int:
        """Delete config"""
        return cls.delete(conn, where={'key': key, 'namespace': namespace})

class GlobalConfigCache:
    """
    Global Configuration Cache Manager (Database-backed)
    """

    DEFAULT_CONFIG = {
        'default_language': 'zh-CN',
        'default_tts_provider': 'edge',
        'default_stt_provider': 'azure',
        'supported_languages': ['zh-CN', 'en-US', 'ja-JP', 'ko-KR', 'lo-LA'],
        'web_interface_enabled': True,
        'clipboard_sync_enabled': True,
        'clipboard_sync_interval': 5,
        'max_clipboard_history': 1000,

        # Speech defaults (NEW)
        'speech_default_device': None,
        'speech_default_languages': ['en-US'],
        'speech_duration_mode': 'continuous',
        'speech_auto_start': False,
        'speech_rpc_port': 59000,
        'speech_provider': 'azure',
        'speech_save_transcripts': True,
        'speech_enable_clipboard': True,
    }

    def __init__(self, use_database: bool = True, namespace: str = 'default'):
        self.use_database = use_database
        self.namespace = namespace
        self._db_manager = None
        self._db_model = None
        self._config: Dict[str, Any] = self.DEFAULT_CONFIG.copy()

        if use_database:
            self._init_database()
        else:
            self._init_json_file()

    def _init_database(self):
        """Initialize database connection"""
        try:
            from pycore.database import database_manager, DATABASE_AVAILABLE
            from pycore.database.models import UtilConfigModel, TableKeys

            if not DATABASE_AVAILABLE:
                ColorPrint.yellow("[GlobalConfig] Database not available, using JSON file")
                self.use_database = False
                self._init_json_file()
                return

            # Register database
            database_manager.register_database("config")

            # Load table
            database_manager.load_tables(
                database_name="config",
                table_keys=[TableKeys.UTIL_CONFIG],
                models=[UtilConfigModel]
            )

            self._db_manager = database_manager
            self._db_model = UtilConfigModel

            # Load configs from database
            self._load_from_database()

            ColorPrint.green(f"[GlobalConfig] Database initialized (namespace: {self.namespace})")

        except Exception as e:
            ColorPrint.yellow(f"[GlobalConfig] Database init failed: {e}, using JSON file")
            self.use_database = False
            self._init_json_file()

    def _load_from_database(self):
        """Load all configs from database"""
        with self._db_manager.get_connection("config") as conn:
            db_configs = self._db_model.get_namespace_configs(conn, self.namespace)
            self._config.update(db_configs)

    def get(self, key: str, default: Any = None) -> Any:
        """Get config value"""
        if self.use_database:
            with self._db_manager.get_connection("config") as conn:
                value = self._db_model.get_config(conn, key, self.namespace)
                return value if value is not None else default
        else:
            return self._config.get(key, default)

    def set(self, key: str, value: Any, description: Optional[str] = None) -> bool:
        """Set config value"""
        try:
            if self.use_database:
                with self._db_manager.get_connection("config") as conn:
                    self._db_model.set_config(conn, key, value, self.namespace, description)

            # Also update in-memory cache
            self._config[key] = value

            # Save to JSON file (backup)
            self._save_json_backup()

            ColorPrint.green(f"[GlobalConfig] Set {key} = {value}")
            return True

        except Exception as e:
            ColorPrint.red(f"[GlobalConfig] Failed to set {key}: {e}")
            return False

    def get_namespace_configs(self, namespace: Optional[str] = None) -> Dict[str, Any]:
        """Get all configs for a namespace"""
        target_namespace = namespace or self.namespace

        if self.use_database:
            with self._db_manager.get_connection("config") as conn:
                return self._db_model.get_namespace_configs(conn, target_namespace)
        else:
            return self._config.copy()
```

```python
import sys

def migrate_config():

    ColorPrint.blue("\n=== Config Migration: JSON → Database ===\n")

    for key, value in old_data.items():

        ColorPrint.green(f"  Migrated: {key}")

    ColorPrint.green(f"\n✅ Migration complete: {len(old_data)} configs")
```

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""TTS RPC Routes"""

from typing import Dict, Any
from pycore.pyfoundations import ColorPrint


def register_tts_routes(rpc_server, service_instances):
    """
    Register TTS routes

    Args:
        rpc_server: ThreadedRpcServer instance
        service_instances: Dict with 'tts_switch', 'stt_switch', etc.
    """

    def handle_tts(params: Dict, request_id: str, context: Dict) -> Dict[str, Any]:
        """
        Handle TTS request - submits to GlobalTaskQueue

        Params:
            text: str - Text to synthesize (required)
            language: str - Language code (default: zh-CN)
            voice: str - Voice name (optional)
            provider: str - TTS provider (default: edge)
            return_base64: bool - Return audio as base64 (default: false)
            enable_cache: bool - Use cache if available (default: true)
            async: bool - Async mode (default: true)
            priority: str - Task priority (HIGH, NORMAL, LOW)

        Returns:
            Async: {success, data: {task_id, status}}
            Sync: {success, data: {audio_file, audio_base64, ...}}
        """
        from pycore.pyfoundations import get_global_task_queue, TaskPriority
        from pycore.pyutils.common.speech_task_models import TTSTaskData, create_tts_task
        import threading

        # Validate params
        text = params.get('text', '').strip()
        if not text:
            return {'success': False, 'error': 'text is required'}

        # Create task data
        task_data = TTSTaskData(
            text=text,
            language=params.get('language', 'zh-CN'),
            voice=params.get('voice'),
            provider=params.get('provider', 'edge'),
            return_base64=params.get('return_base64', False),
            enable_cache=params.get('enable_cache', True),
            request_id=request_id,
            client_id=context.get('client_id', 'unknown')
        )

        # Map priority
        priority_map = {
            'CRITICAL': TaskPriority.CRITICAL,
            'HIGH': TaskPriority.HIGH,
            'NORMAL': TaskPriority.NORMAL,
            'LOW': TaskPriority.LOW
        }
        priority = priority_map.get(params.get('priority', 'NORMAL').upper(), TaskPriority.NORMAL)

        # Create task
        task = create_tts_task(task_data, priority=priority)

        # Submit to global queue
        task_queue = get_global_task_queue()
        task_queue.put(task)

        # Async mode - return task ID
        if params.get('async', True):
            return {
                'success': True,
                'data': {
                    'task_id': task.task_id,
                    'status': 'submitted',
                    'estimated_wait_time': task_queue.size() * 0.5
                }
            }

        # Sync mode - wait for completion
        completed_event = threading.Event()
        task_result = {'result': None, 'error': None}

        def on_complete(t):
            task_result['result'] = t.metadata.get('result', {})
            completed_event.set()

        def on_error(t):
            task_result['error'] = t.error or "Task failed"
            completed_event.set()

        task.callback = on_complete
        task.error_callback = on_error

        # Wait for completion (max 30s)
        completed_event.wait(timeout=30.0)

        if task_result['error']:
            return {'success': False, 'error': task_result['error']}

        if task_result['result']:
            return {'success': True, 'data': task_result['result'].get('data', {})}

        return {'success': False, 'error': 'Task timeout'}

    # Register routes
    rpc_server.route('tts', handle_tts)
    rpc_server.route('tts.synthesize', handle_tts)

    ColorPrint.green("[TTS Routes] Registered")
```

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""Queue Management RPC Routes"""

from typing import Dict, Any
from pycore.pyfoundations import ColorPrint


def register_queue_routes(rpc_server, service_instances):
    """Register queue management routes"""

    def handle_queue_stats(params: Dict, request_id: str, context: Dict) -> Dict[str, Any]:
        """Get queue statistics"""
        from pycore.pyheartbeat import get_heartbeat_system

        heartbeat_system = get_heartbeat_system()
        stats = heartbeat_system.get_stats()

        return {
            'success': True,
            'data': stats
        }

    def handle_task_status(params: Dict, request_id: str, context: Dict) -> Dict[str, Any]:
        """
        Get task status by task_id

        Params:
            task_id: str - Task ID to query

        Returns:
            {success, data: {task_id, type, status, result, ...}}
        """
        task_id = params.get('task_id')

        if not task_id:
            return {'success': False, 'error': 'task_id is required'}

        # Query task from global queue
        from pycore.pyfoundations import get_global_task_queue
        task_queue = get_global_task_queue()

        # Find task in queue
        task = None
        for t in task_queue._queue:
            if isinstance(t, tuple):
                if t[1].task_id == task_id:
                    task = t[1]
                    break
            elif hasattr(t, 'task_id') and t.task_id == task_id:
                task = t
                break

        if not task:
            return {'success': False, 'error': 'Task not found'}

        return {
            'success': True,
            'data': {
                'task_id': task.task_id,
                'type': task.task_type,
                'status': task.state.value,
                'priority': task.priority.value,
                'created_at': task.created_at,
                'started_at': task.started_at,
                'completed_at': task.completed_at,
                'duration': task.get_duration(),
                'error': task.error,
                'result': task.metadata.get('result')
            }
        }

    # Register routes
    rpc_server.route('queue_stats', handle_queue_stats)
    rpc_server.route('task_status', handle_task_status)

    ColorPrint.green("[Queue Routes] Registered")
```

### 3.3 Simplified rpc_service.py

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
RPC Service - Thread-based RPC Service for Speech (Simplified)

Uses modular route registration from routes/ directory.
"""

import threading
from typing import Optional
from pycore.pyfoundations import ColorPrint

# Import all route registration functions
from pycore.pyctl.speech.rpc.routes import (
    register_tts_routes,
    register_stt_routes,
    register_config_routes,
    register_status_routes,
    register_queue_routes,
    register_clipboard_routes,
    register_transcript_routes
)


class RPCService:
    """
    RPC Service - Registers speech routes on ThreadedRpcServer

    Simplified version - routes are in separate files
    """

    def __init__(self, rpc_server, tts_switch=None, stt_switch=None):
        """
        Initialize RPC Service

        Args:
            rpc_server: ThreadedRpcServer instance (already running as thread)
            tts_switch: TTSSwitch instance (optional)
            stt_switch: STTSwitch instance (optional)
        """
        self.server = rpc_server
        self.service_instances = {
            'tts_switch': tts_switch,
            'stt_switch': stt_switch
        }
        self._registered = False

    def register_routes(self):
        """Register all speech-related routes"""
        if self._registered:
            ColorPrint.yellow("[RPCService] Routes already registered")
            return

        ColorPrint.blue("[RPCService] Registering speech routes...")

        # Register modular routes
        register_tts_routes(self.server, self.service_instances)
        register_stt_routes(self.server, self.service_instances)
        register_config_routes(self.server, self.service_instances)
        register_status_routes(self.server, self.service_instances)
        register_queue_routes(self.server, self.service_instances)
        register_clipboard_routes(self.server, self.service_instances)
        register_transcript_routes(self.server, self.service_instances)

        self._registered = True
        ColorPrint.green("[RPCService] All routes registered successfully")


# Global singleton
_global_rpc_service: Optional[RPCService] = None
_service_lock = threading.Lock()


def get_rpc_service(rpc_server=None, tts_switch=None, stt_switch=None) -> RPCService:
    """Get global RPC service singleton"""
    global _global_rpc_service
    with _service_lock:
        if _global_rpc_service is None:
            if rpc_server is None:
                raise ValueError("rpc_server required for first initialization")
            _global_rpc_service = RPCService(rpc_server, tts_switch, stt_switch)
        return _global_rpc_service


def start_rpc_service(rpc_server, tts_switch=None, stt_switch=None) -> RPCService:
    """
    Register speech routes on running RPC server

    Args:
        rpc_server: ThreadedRpcServer instance (already running)
        tts_switch: TTSSwitch instance (optional)
        stt_switch: STTSwitch instance (optional)

    Returns:
        RPCService instance
    """
    service = get_rpc_service(rpc_server, tts_switch, stt_switch)
    service.register_routes()
    return service


__all__ = [
    'RPCService',
    'get_rpc_service',
    'start_rpc_service'
]
```

```python
def accept_task(self, task: Task) -> bool:
    """Accept task from HeartbeatPusher"""
    if self.task_queue.qsize() < self.max_queue_size:
        try:
            self.task_queue.put(task, block=False)
            task.mark_running()
            return True
        except:
            return False
    return False

def _process_task(self, task: Task):
    """Process TTS task"""
    from pycore.pyutils.common.speech_task_models import TTSTaskData, TTSTaskResult

    # Parse task data
    task_data = TTSTaskData.from_dict(task.task_data)

    # Route to provider (edge/azure)
    result = self._route_to_provider(task_data)

    # Store result in task metadata
    task.metadata['result'] = {'data': result.to_dict()}
    task.mark_completed()

    # Call callback if set
    if task.callback:
        task.callback(task)
```

```javascript
function startTaskPolling(taskId, taskType, callbacks) {
    const { onUpdate, onComplete, onError } = callbacks;
    const startTime = Date.now();

    const pollTask = async () => {
        try {
            const statusResult = await rpcClient.call('task_status', { task_id: taskId });

            if (!statusResult.success) {
                stopTaskPolling(taskId);
                if (onError) onError(statusResult.error || 'Failed to get task status');
                return;
            }

            const status = statusResult.data;

            if (onUpdate) onUpdate(status);

            if (status.status === 'completed') {
                stopTaskPolling(taskId);
                if (onComplete) onComplete(status);
            } else if (status.status === 'failed') {
                stopTaskPolling(taskId);
                if (onError) onError(status.error || 'Task failed');
            }
            // Otherwise continue polling (pending/running)

        } catch (error) {
            stopTaskPolling(taskId);
            if (onError) onError(error.message || 'Polling error');
        }
    };

    // Poll every 1 second
    const intervalId = setInterval(pollTask, 1000);

    activeTasks[taskId] = {
        type: taskType,
        startTime: startTime,
        pollInterval: intervalId
    };

    // Initial poll
    pollTask();
}
```

```python
# pycore/pylauncher/launcher.py
def find_available_port(start_port: int = 59000, range_size: int = 100) -> int:
    """
    Find available port in range

    Args:
        start_port: Starting port (default: 59000)
        range_size: Port range size (default: 100)

    Returns:
        Available port number

    Raises:
        RuntimeError: If no ports available
    """
    import socket

    for port in range(start_port, start_port + range_size):
        try:
            sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            sock.bind(('0.0.0.0', port))
            sock.close()
            return port
        except OSError:
            continue

    raise RuntimeError(f"No available ports in range {start_port}-{start_port+range_size}")


# Use in launch_speech_only
```
