SHARED FEATURES:
---------------
✓ Common configuration for both HTTP and WebSocket RPC
✓ Same port for HTTP and WebSocket (USE_SAME_PORT: true)
✓ Shared client cache using encyclopedia.js
✓ Unified constants and error codes
✓ Consistent API across modules

USAGE EXAMPLES:
--------------

1. Unified Import:
   const rpc = require('#@ncore/utils/rpc');

   // WebSocket RPC
   const wsServer = rpc.createWsServer({ port: 8080 });
   const wsClient = rpc.createWsClient('ws://localhost:8080');

   // HTTP RPC
   const httpServer = rpc.createHttpServer(expressApp);
   const httpClient = rpc.createHttpClient('http://localhost:8080/rpc');

   // Express Server
   const server = await rpc.startExpressServer({ HTTP_PORT: 8080 });

2. Configuration:
   const { setConfig, getConfig } = require('#@ncore/utils/rpc');

   setConfig({
       HTTP_PORT: 8080,
       WS_PORT: 8080,
       USE_SAME_PORT: true,
       AUTH_ENABLED: true
   });

   const config = getConfig();

3. Client Cache:
   const { getCache } = require('#@ncore/utils/rpc');

   const cache = getCache();
   cache.set('user:123', { name: 'John' }, 3600000);
   const user = cache.get('user:123');

4. WebSocket RPC:
   const { ws } = require('#@ncore/utils/rpc');

   const server = ws.createServer();
   server.route('getData', async (params) => {
       return { data: 'result' };
   });
   await server.start();

5. HTTP RPC + Express:
   const { http } = require('#@ncore/utils/rpc');

   const server = http.createExpressServer({ HTTP_PORT: 8080 });

   server.getRouterManager().get('/api/test', (req, res) => {
       res.json({ status: 'ok' });
   });

   await server.start();

6. Constants:
   const { RPC_CONSTANTS, MSG_TYPES, ERROR_CODES } = require('#@ncore/utils/rpc');

   console.log(MSG_TYPES.REQUEST);    // 'request'
   console.log(ERROR_CODES.TIMEOUT);  // 'TIMEOUT'
